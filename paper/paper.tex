\title{On Ordowski's characterization of odd primes by the number of squares modulo $n$ (OEIS A000224): even numbers, prime powers, numbers $p^aq^b$, and products of three distinct primes, formalized in Lean~4}
\begin{document}
\maketitle
\begin{abstract}
Let $a(n)$ be the number of squares modulo $n$ (OEIS A000224~\cite{oeis}). T.~Ordowski conjectured that for $n>1$, $n^2\equiv1\pmod{a(n)(a(n)-1)}$ holds if and only if $n$ is an odd prime. \textbf{The full conjecture remains open.} We prove it for every $n>1$ that is even, a prime power $p^k$ ($k\ge1$), a number $p^aq^b$ with $p\ne q$ odd primes and $a,b\ge1$ (in particular every $pq$), or a product $pqr$ of three distinct odd primes. The prime-power case uses Stangl's formula for $a(p^k)$~\cite{stangl}, which we also formalize. The semiprime case combines a size bound with a divisibility bound and a finite check. The three-prime case bounds the smallest prime by a size argument, views the congruence as a quadratic in the largest prime to bound the middle one, and finishes with a kernel-checked finite verification of about $9.4\times10^4$ cases. The case $p^aq^b$ uses Stangl's formula to write the congruence as a quadratic in $p^a$ and in $q^b$; size bounds give $Q\le40$ and $p\le67$, explicit bounds on $q$ and on $\min(p^a,q^b)$ follow, and a kernel-checked integer-root search finishes the proof. Everything is formalized in Lean~4 with Mathlib against the exact definition of $a$ in Google DeepMind's \texttt{formal-conjectures} repository~\cite{fc}. The development uses no \texttt{sorry} and no \texttt{native\_decide}, and only the standard axioms.
\end{abstract}

\section{Introduction}
For $n\ge1$ let $a(n)=\#\{k^2\bmod n:0\le k<n\}$.
\begin{conjecture}[Ordowski, 2025]\label{conj}
For $n>1$, $n$ is an odd prime if and only if $n^2\equiv1\pmod{a(n)(a(n)-1)}$.
\end{conjecture}
The conjecture appears as a comment on A000224~\cite{oeis}, together with M.~Marcus's remark that it holds up to $n=10^8$. It is formalized as \lean{OeisA224.conjecture} in \nolinkurl{FormalConjectures/OEIS/224.lean} of \texttt{formal-conjectures}~\cite{fc}, where it is tagged \texttt{research open}. The forward direction is immediate from $a(p)=(p+1)/2$. The open part is to show that no composite $n$ satisfies the congruence.

\section{Related work}
The count $a(n)$ is multiplicative, and Stangl~\cite{stangl} gives its value at prime powers; see also Finch and Sebah~\cite{fs}. On October 12, 2026 we searched OEIS, the web, MathOverflow and the issues and pull requests of \texttt{formal-conjectures} (upstream commit \texttt{34ee0e3}). We found no proof of Conjecture~\ref{conj}, partial or complete. The cases treated here therefore appear to be new, although the arguments are elementary. The formalization uses Lean~4~\cite{lean4} and Mathlib~\cite{mathlib}.

\section{Statement}
\begin{theorem}\label{main}
Conjecture~\ref{conj} holds for every $n>1$ that is (i) even, (ii) a prime power $p^k$ with $k\ge1$, or (iii) of the form $p^aq^b$ with $p\ne q$ odd primes and $a,b\ge1$ (this includes $pq$), or (iv) a product $pqr$ of distinct odd primes $p,q,r$.
\end{theorem}
\noindent The cases not covered are exactly the odd $n$ with at least three distinct prime factors, other than squarefree $pqr$ (for example $pqrs$ or $p^2qr$).

\section{Proof overview}
\subsection{Preliminaries}
\begin{lemma}\label{mult}
$a(n)=\#\{x^2 : x\in\mathbb{Z}/n\}$; $a$ is multiplicative; and $a(p)=(p+1)/2$ for odd primes $p$.
\end{lemma}
\begin{proof}
The first claim holds because reduction mod $n$ is a bijection $\{0,\dots,n-1\}\to\mathbb{Z}/n$. If $\gcd(m,n)=1$, the CRT ring isomorphism $\mathbb{Z}/mn\cong \mathbb{Z}/m\times\mathbb{Z}/n$ commutes with squaring, so it maps the set of squares onto the product of the two sets of squares. For an odd prime $p$, the squares are the values $k^2$ with $0\le k\le (p-1)/2$, since $x^2=(-x)^2$. These values are distinct: $k^2=l^2$ forces $p\mid (k-l)(k+l)$, and $0\le k+l<p$.
\end{proof}

\subsection{The even case}
If $n$ is even, then $n^2-1$ is odd, while $a(n)(a(n)-1)$ is a product of two consecutive integers and hence even. So $n^2\not\equiv 1$. (If $a(n)(a(n)-1)=0$, the congruence would say $n^2=1$, which is impossible for $n>1$.)

\subsection{The semiprime case}
\begin{theorem}\label{pq}
Let $p<q$ be odd integers with $p\ge 3$ prime, and let $s=(p+1)(q+1)/4$. Then $s(s-1)\nmid (pq)^2-1$.
\end{theorem}
Primality of $q$ is not needed. By Lemma~\ref{mult}, $a(pq)=s$ for distinct odd primes, so Theorem~\ref{pq} gives the semiprime case of Theorem~\ref{main}.
\begin{proof}
Put $P=(p+1)(q+1)$, so that $16\,s(s-1)=P(P-4)$. Suppose $Q\,s(s-1)=(pq)^2-1$, so $Q\,P(P-4)=16((pq)^2-1)$.

\emph{Step 1: $Q\le 15$.} Expanding, $P(P-4)-((pq)^2-1)=2pq(p+q+1)-4pq+(p+q+1)^2-4(p+q+1)+1>0$. Hence $Q<16$.

\emph{Step 2: $p\le 61$.} By Step~1, $16((pq)^2-1)\le 15P(P-4)<15P^2$. Suppose $p\ge 62$. Since $q>p$ we have $(q+1)p\le (p+1)q$, so $P^2p^2\le (p+1)^4q^2$. Also $15(p+1)^4\le 16p^4-16$ for $p\ge 62$. Together with $q\ge p$ this gives $15P^2\le 16((pq)^2-1)$, a contradiction.

\emph{Step 3: $q+1\mid 16(p^2-1)$.} Since $q+1\mid P$, we get $q+1\mid 16((pq)^2-1)$. Also $16((pq)^2-1)-16(p^2-1)=16p^2(q^2-1)$ is divisible by $q+1$.

\emph{Step 4: finite check.} For each prime $p\in[3,61]$ and each divisor $d=q+1$ of $16(p^2-1)$ with $q>p$ odd, direct computation shows $s(s-1)\nmid (pq)^2-1$. There are only finitely many such pairs.
\end{proof}

\subsection{Three distinct odd primes}
\begin{theorem}\label{pqr}
Let $3\le p<q<r$ be integers with $p$ prime, and let $s$ satisfy $8s=(p+1)(q+1)(r+1)$. Then $s(s-1)\nmid (pqr)^2-1$.
\end{theorem}
Primality of $q$ and $r$ is not needed. By Lemma~\ref{mult}, $a(pqr)=s$ for distinct odd primes, which gives case (iv) of Theorem~\ref{main}.
\begin{proof}
Let $n=pqr$ and $P=(p+1)(q+1)(r+1)=8s$, and suppose $Q\,s(s-1)=n^2-1$. Then $Q\,P(P-8)=64(n^2-1)$.

\emph{Step 1: $1\le Q\le 63$.} Since $P-8\ge n$, we have $P(P-8)>n^2-1>0$.

\emph{Step 2: $p\le 400$.} From $p\le q\le r$ we get $p^3P\le (p+1)^3n$. Combined with $64(n^2-1)\le 63P^2$ this gives $\bigl(64p^6-63(p+1)^6\bigr)n^2\le 64p^6$. For $p\ge 401$ the bracket is at least $65$, and $n^2>p^6$, a contradiction.

\emph{Step 3: a quadratic in $r$.} Put $A=(p+1)(q+1)$ and $B=pq$. Expanding the identity gives
\[(64B^2-QA^2)\,r^2=(2QA^2-8QA)\,r+(QA^2-8QA+64),\]
where both coefficients on the right are positive. Hence $c_2:=64B^2-QA^2>0$. With $K=Q(p+1)^2$ and $d=64p^2-K$ we have $c_2=dq^2-K(2q+1)$, so $d>0$. Moreover $c_2r\le c_1+c_0<3QA^2$ and $q<r$, so $c_2q<3K(q+1)^2$. If $q\ge 8$ and $dq\ge 6K$, then $c_2q\ge 4Kq^2-Kq>3K(q+1)^2$, a contradiction. Therefore $q\le 6K/d+8$. The near-misses $Q(1+1/p)^2=64$, namely $(p,Q)=(3,36)$ and $(7,49)$, have $d=0$ and are excluded.

\emph{Step 4: finite check.} For each prime $p\le 400$, each $Q\le 63$ with $d>0$, and each $q\le 6K/d+8$ with $q>p$ (about $9.4\times10^4$ triples), the quadratic in Step~3 has no natural root $r>q$. A root would equal $(c_1+c)/(2c_2)$, where $c=\sqrt{c_1^2+4c_2c_0}$. The check proposes $c$ by Newton's method and verifies $c^2\le D<(c+1)^2$ directly, so the correctness of Newton's method is never used.
\end{proof}

\subsection{Odd prime powers}
\begin{lemma}[Stangl~\cite{stangl}]\label{stangl}
For an odd prime $p$ and $k\ge 2$, $a(p^k)=a(p^{k-2})+\tfrac12 p^{k-1}(p-1)$. Consequently $2(p+1)\,a(p^k)=p^{k+1}+p+2$ if $k$ is even and $p^{k+1}+2p+1$ if $k$ is odd.
\end{lemma}
\begin{proof}
Let $N=p^k$. A square $x^2 \bmod N$ is divisible by $p$ iff $p\mid x$. Writing $x=pt$ gives $(pt)^2 \bmod p^2M=p^2(t^2\bmod M)$ with $M=p^{k-2}$, so the squares divisible by $p$ are exactly $p^2$ times the squares modulo $M$. For the squares prime to $p$: if $x,y\in[0,N/2)$ are prime to $p$ and $N\mid (y-x)(y+x)$, then $p$ cannot divide both factors (as $p$ is odd and $p\nmid y$), so $N$ divides one factor, forcing $x=y$. Replacing $x$ by $N-x$, every unit square has such a representative. Hence the unit squares number $\varphi(N)/2$. The closed form follows by induction from $a(1)=1$ and $a(p)=(p+1)/2$.
\end{proof}
\begin{theorem}\label{pp}
If $p$ is an odd prime, $k\ge 2$, and $s$ satisfies the identity of Lemma~\ref{stangl}, then $s(s-1)\nmid p^{2k}-1$. Consequently, $(p^k)^2\not\equiv 1\pmod{a(p^k)(a(p^k)-1)}$, and Conjecture~\ref{conj} holds for all odd prime powers.
\end{theorem}
\begin{proof}[Sketch]
Let $X=p^k$ and $M=2(p+1)$, and write $Q\,s(s-1)=X^2-1$. Multiplying by $M^2$ gives $Q(Ms)(Ms-M)=M^2(X^2-1)$. For even $k$, reducing modulo $p$ gives $p\mid 4(p+1)^2$, which is impossible. For odd $k\ge 3$, a size estimate gives $1\le Q\le 7$, and reducing modulo $p$ gives $p\mid Q-4$. If $Q=4$, the equation becomes $(2p+1)X^2=2p^2X+p^2$, which is impossible since $X>p$. Otherwise $p=3$ and $Q\in\{1,7\}$, and both cases are ruled out by elementary inequalities.
\end{proof}

\subsection{Two distinct primes with arbitrary exponents}
\begin{theorem}\label{ppq}
Let $p<q$ be distinct odd primes and $a,b\ge1$. Then $n=p^aq^b$ does not satisfy $n^2\equiv1\pmod{a(n)(a(n)-1)}$.
\end{theorem}
\begin{proof}[Sketch]
Write $X=p^a$, $Y=q^b$, $s=a(n)=a(X)a(Y)$ and $M=4(p+1)(q+1)$. By Lemma~\ref{stangl}, $U:=2(p+1)a(X)=pX+c$ and $V:=2(q+1)a(Y)=qY+e$ with $c\in\{p+2,2p+1\}$ and $e\in\{q+2,2q+1\}$, so $Ms=UV$. Suppose $Q\,s(s-1)=n^2-1$. Then $Q$ is prime to $n$ (it divides $n^2-1$), and
\[Q\,UV(UV-M)=M^2(X^2Y^2-1). \tag{$*$}\]

\emph{Step 1: $1\le Q\le 40$.} Expanding, $UV-M\ge pq\,n$ (the term $qYc\ge q^2(p+2)$ already exceeds $M$), so $Q\,p^2q^2n^2\le M^2(n^2-1)$ and $Qp^2q^2<M^2=16(p+1)^2(q+1)^2$. Since $8pq\ge5(p+1)(q+1)$ for $p\ge3$, $q\ge5$, this forces $Q\le40$.

\emph{Step 2: $p\le67$.} If $p\ge71$ then $pq/((p+1)(q+1))\ge 5112/5256$, and comparing $(*)$ with $pqn\le Ms\le ((p+1)(q+1))^2n/(pq)$ gives $15<Q<17$. But $Q=16$ means $(4s-2)^2=n^2+3$, impossible for $n\ge2$. The primes $68\le p\le70$ do not exist.

\emph{Step 3: two quadratics.} Substituting $U=pX+c$ in $(*)$ gives
\[(M^2Y^2-Qp^2V^2)X^2=QpV(2cV-M)\,X+\bigl(QcV(cV-M)+M^2\bigr),\]
and symmetrically a quadratic in $Y$ with leading coefficient $M^2X^2-Qq^2U^2$. Because $cV>M$ and $eU>M$, both right-hand sides have positive coefficients, so both leading coefficients are positive.

\emph{Step 4: bounding $q$.} Let $E=16(p+1)^2-Qp^2$ (never $0$, since $p\nmid16(p+1)^2$) and $G=16(p+1)^2X^2-QU^2$. The quadratic in $Y$ together with $Y\ge q$ gives $q(-G)<48(p+1)^2X^2$ if $G<0$ and $qG\le13QU^2$ if $G>0$. If $E<0$ this yields $q\le 48(p+1)^2/(-E)$. If $E>0$ and $2Q(2p+1)(2pX+2p+1)\le EX^2$ (automatic for $X>2\cdot10^6$), then $2G\ge EX^2$ and $pU\le(p+1)^2X$ give $q\le 26Q(p+1)^4/(p^2E)$. The finitely many remaining $X\le2\cdot10^6$ are treated one at a time with the $X$-dependent bound.

\emph{Step 5: bounding $\min(X,Y)$.} Write $\Delta=M^2-Qp^2q^2>0$. Expanding $(*)$ with $U^2-p^2X^2\le9p^2X$ and $V^2-q^2Y^2\le9q^2Y$ gives $\min(X,Y)\,\Delta<45Qp^2q^2+M^2$.

\emph{Step 6: finite check.} For every prime $p\le67$, every $1\le Q\le40$ with $p\nmid Q$, every odd $q>p$ under the bound of Step~4, and every power $X=p^a$ (resp.\ $Y=q^b$) under the bound of Step~5 (or $X\le2\cdot10^6$ in the small-$X$ branch), and for both admissible values of $e$ (resp.\ $c$), the quadratic in the other variable has no natural root divisible by $q$ (resp.\ $p$). Roots are tested as in Theorem~\ref{pqr}: a Newton candidate $c$ is verified by $c^2\le D<(c+1)^2$.
\end{proof}
\noindent The case $a=b=1$ is covered as well, so Theorem~\ref{ppq} contains the semiprime case. An independent search up to $n\le10^{15}$ also found no counterexample of this shape.

\section{Limitations}
\textbf{The full conjecture remains open.} The untreated case is odd $n$ with at least three distinct prime factors, other than squarefree $pqr$; for example $pqrs$ and $p^2qr$. For squarefree $n$ with $k$ factors, the argument of Theorem~\ref{pqr} gives $Q<4^k$ and a bound on the smallest prime of order $k\,4^k$. For each fixed $k$ an iterated version of Steps~2--4 reduces the problem to a finite search, but the search grows far too quickly to check in the kernel even for $k=4$. A proof uniform in $k$ would need a new idea. The independent computational checks in \texttt{scripts/} are not part of the formal proof.
